CREATE PROCEDURE PatientReport
AS
Select 'Dr.' + D.surname as 'DOCTOR.surname', 
P.surname as 
'PATIENT.surname' , p.forename, appointmentdate
From Appointment A
Inner join Doctor D
On A.StaffNo=D.StaffNo
Inner join Patient P
On A.PatientNo=P.PatientNo
Order by D.Surname,P.Surname,AppointmentDate desc

GO;
